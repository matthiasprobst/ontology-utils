= Provenance with PROV

A project can use PROV terms for its own provenance model.

[source,python]
----
from ontolutils import Thing, namespaces, urirefs
from ontolutils.typing import AnyIri


@namespaces(prov="http://www.w3.org/ns/prov#")
@urirefs(Entity="prov:Entity", attribution="prov:wasAttributedTo")
class Entity(Thing):
    attribution: AnyIri


dataset = Entity(
    id="https://example.org/datasets/run-01",
    attribution="https://example.org/people/ada",
)
assert "prov:wasAttributedTo" in dataset.serialize("ttl")
----
