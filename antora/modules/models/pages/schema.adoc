= Software metadata with Schema.org

[source,python]
----
from ontolutils import Thing, namespaces, urirefs
from ontolutils.typing import AnyIri


@namespaces(schema="https://schema.org/")
@urirefs(SoftwareSourceCode="schema:SoftwareSourceCode", repository="schema:codeRepository")
class SoftwareSourceCode(Thing):
    repository: AnyIri


software = SoftwareSourceCode(
    id="https://example.org/software/velocity",
    repository="https://example.org/code",
)
assert "schema:codeRepository" in software.serialize("ttl")
----
